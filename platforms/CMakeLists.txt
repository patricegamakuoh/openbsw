if (NOT BUILD_UNIT_TESTS)
    # Allow selecting platform explicitly:
    #   -DOPENBSW_PLATFORM=s32k1xx | s32k3xx | posix
    set(OPENBSW_PLATFORM ${OPENBSW_PLATFORM} CACHE STRING "Target platform (s32k1xx|s32k3xx|posix)")

    if (NOT OPENBSW_PLATFORM)
        # Try to auto-detect from common CPU/platform defines if caller didn't specify
        if (DEFINED S32K3XX OR DEFINED CPU_S32K312 OR DEFINED CPU_S32K344 OR DEFINED CORE_M7)
            set(OPENBSW_PLATFORM s32k3xx)
        elseif (DEFINED S32K1XX OR DEFINED CPU_S32K148 OR DEFINED CORE_CM4 OR DEFINED CORE_M4)
            set(OPENBSW_PLATFORM s32k1xx)
        else ()
            message(FATAL_ERROR "Please set -DOPENBSW_PLATFORM to one of: s32k1xx, s32k3xx, posix")
        endif ()
    endif ()

    message(STATUS "OpenBSW platform: ${OPENBSW_PLATFORM}")

    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${OPENBSW_PLATFORM}/CMakeLists.txt)
        add_subdirectory(${OPENBSW_PLATFORM})
    else ()
        message(FATAL_ERROR "Unknown platform path or missing CMakeLists.txt: ${OPENBSW_PLATFORM}")
    endif ()
else ()
    add_subdirectory(posix/bsp/socketCanTransceiver)
    add_subdirectory(posix/bsp/socketCanTransceiver/test/gtest)
    add_subdirectory(posix/bsp/bspEepromDriver/test/gtest)
    
    # S32K1xx unit tests
    add_subdirectory(s32k1xx/bsp/bspCore/test/gtest)
    add_subdirectory(s32k1xx/bsp/bspEepromDriver/test/gtest)
    add_subdirectory(s32k1xx/bsp/bspIo/test/gtest)
    add_subdirectory(s32k1xx/bsp/bspSci/test/gtest)
    add_subdirectory(s32k1xx/bsp/canflex2Transceiver)
    add_subdirectory(s32k1xx/bsp/canflex2Transceiver/test/gtest)
    add_subdirectory(s32k1xx/bsp/canflex2Transceiver/test/mock)

    # S32K3xx unit tests (only add if paths exist)
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/bspCore/test/gtest/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/bspCore/test/gtest)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/bspEepromDriver/test/gtest/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/bspEepromDriver/test/gtest)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/bspIo/test/gtest/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/bspIo/test/gtest)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/bspSci/test/gtest/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/bspSci/test/gtest)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/canflex2Transceiver/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/canflex2Transceiver)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/canflex2Transceiver/test/gtest/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/canflex2Transceiver/test/gtest)
    endif ()
    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/s32k3xx/bsp/canflex2Transceiver/test/mock/CMakeLists.txt)
        add_subdirectory(s32k3xx/bsp/canflex2Transceiver/test/mock)
    endif ()
endif ()
